\section*{\centering Вариант 1}

\begin{enumerate}
    \item Найти линию, проходящую через точку $(6, 4)$ и обладающую тем свойством,
    что в любой её точке $M$ нормальный вектор $\overrightarrow{MN}$ с концом на оси $OY$ имеет длину,
    равную 10, и образует острый угол с положительным направлением оси $OY$.
    \item Найти решение уравнения, ограниченное при $x \rightarrow \infty$
    $$
        y' - y \cdot ln(2) = 2^{sin(x)}(cos(x) - 1) \cdot ln(2)
    $$
    \item Найти общий интеграл уравнения
    $$
        \dfrac{1 + xy}{x^2y}dx + \dfrac{1-xy}{xy^2}dy = 0
    $$
    \item Найти общее решение уравнения и определить, имеет ли оно особые решения
    $$
        (y')^2 - 2xy' + y = 0
    $$
    \item Найти решение задачи Коши
    $$
        y'' = \dfrac{y'}{x}\left(1 + \dfrac{y'}{x}\right),\ y(1) = \dfrac{1}{2},\ y'(1) = 1.
    $$
\end{enumerate}

\section*{\centering Вариант 2}

\begin{enumerate}
    \item Найти линию, проходящую через точку $(-1, 1)$,
    если отрезок любой её касательной между точкой касания и осью $OY$ делится в точке пересечения с осью абсцисс в отношении $3:1$,
    считая от оси $OY$.
    \item Найти общее решение уравнения
    $$
        y' = \dfrac{2x}{x^2cos(y) + a \cdot sin(2y)}
    $$
    \item Найти общий интеграл уравнения
    $$
        x e^{y^2}dx + (x^2ye^{y^2} + tg^2y)dy = 0
    $$
    \item Найти особое решение уравнения
    $$
        y = x + 2y' - (y')^2
    $$
    \item Найти решение задачи Коши
    $$
        y'y^2 + yy'' - (y')^2 = 0,\ y(0) = 1,\ y'(0) = 2
    $$
\end{enumerate}

\section*{\centering Вариант 3}

\begin{enumerate}
    \item Найти кривые, у которых отрезок, отсекаемый нормалью на оси $OX$, равен $y^2 / x$.
    \item Найти решение уравнения, удовлетворяющее условию: $y \rightarrow 2$ при $x \rightarrow -\infty$
    $$
        y' - e^xy = \dfrac{1}{x^2}sin\dfrac{1}{x} - e^x cos\dfrac{1}{x}
    $$
    \item Найти общий интеграл уравнения
    $$
        \left(\dfrac{x}{sin(y)} + 2\right)dx + \dfrac{(x^2 + y) \cdot cos(y)}{cos(2y) - 1}dy = 0
    $$
    \item Найти особое решение и выделить особые решения, если они существуют
    $$
        8(y')^3 - 12(y')^2 = 27(y - x)
    $$
    \item Найти решение задачи Коши
    $$
        2y' + ((y')^2 - 6x)y'' = 0,\ y(1) = \sqrt{2},\ y'(1) = \sqrt{2}
    $$
\end{enumerate}
